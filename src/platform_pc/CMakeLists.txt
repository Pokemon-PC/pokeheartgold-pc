# PC hardware abstraction layer + host executable.
#
# Milestone 1: the HAL builds on its own and the executable opens the window,
# mounts the NitroFS data directory and renders data read from it.
# Milestone 2+ adds the game sources (src/, lib/NitroSDK, lib/NitroSystem)
# on top of this HAL - see PORTING_ROADMAP.md.

# Data layers are looked up under the repository root at run time:
# mods/<lang>, mods, pc_data/<lang> (e.g. extracted from your own ROM with
# tools/pc_port/nitrofs.py), then files/ (repository sources).
option(PHG_PC_CONSOLE "Keep a console window for logs on Windows" ON)

add_library(phg_platform STATIC
    os_pc.c
    fs_pc.c
    input_pc.c
    config_pc.c
    install_pc.c
    save_pc.c
    gfx_pc.c
    png_pc.c
    audio_pc.c
    log_pc.c
    hw_pc.c
    crash_pc.c
)
target_include_directories(phg_platform PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(phg_platform PUBLIC phg::SDL2 OpenGL::GL)
target_compile_definitions(phg_platform PUBLIC PLATFORM_PC=1 PHG_REPO_DIR="${PROJECT_SOURCE_DIR}"
    ${PHG_GAME_VERSION})   # names, ROM code, data folder of the version (platform_pc.h)

if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(phg_platform PRIVATE -Wall -Wextra -Wno-unused-parameter)
elseif(MSVC)
    target_compile_options(phg_platform PRIVATE /W3)
    target_compile_definitions(phg_platform PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()

# NitroFS manifest: every path the ROM's file system contains, extracted from
# filesystem.mk, so `--fs-check` can report what is missing on disk.
# (Lines end with a '\' continuation, which would escape CMake's list ';'
# separator, so match the paths themselves rather than splitting lines.)
file(READ ${PROJECT_SOURCE_DIR}/filesystem.mk _phg_fs_mk)
string(REGEX MATCH "NITROFS_FILES :=[^\n]*\n(\t[^\n]*\n)*" _phg_fs_block "${_phg_fs_mk}")
string(REGEX MATCHALL "\tfiles/[^ \t\r\n\\\\]+" _phg_fs_paths "${_phg_fs_block}")
set(_phg_manifest "")
foreach(_path IN LISTS _phg_fs_paths)
    string(REGEX REPLACE "^\tfiles/" "" _path "${_path}")
    string(APPEND _phg_manifest "${_path}\n")
endforeach()
file(WRITE ${CMAKE_BINARY_DIR}/nitrofs_manifest.txt "${_phg_manifest}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${PROJECT_SOURCE_DIR}/filesystem.mk)

add_executable(pokeheartgold_pc main_pc.c)
target_link_libraries(pokeheartgold_pc PRIVATE phg::SDL2main phg_platform)
target_compile_definitions(pokeheartgold_pc PRIVATE
    PHG_DEFAULT_MANIFEST="${CMAKE_BINARY_DIR}/nitrofs_manifest.txt")
if(WIN32 AND NOT PHG_PC_CONSOLE)
    set_target_properties(pokeheartgold_pc PROPERTIES WIN32_EXECUTABLE ON)
endif()
if(MINGW)
    # Ship a self-contained .exe (no libgcc/libwinpthread DLLs needed).
    target_link_options(pokeheartgold_pc PRIVATE -static)
endif()
set_target_properties(pokeheartgold_pc PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
phg_low_address_link(pokeheartgold_pc)

# --- The game (milestone 3) ----------------------------------------------------
include(PhgGameSources)
# DS assembly data converted to C by tools/pc_port/asm_data_to_c.py.
# Checked-in generated sources, see src/platform_pc/generated/README.md:
# DS assembly data converted to C, and offline stubs for online / unsupported
# hardware libraries.
file(GLOB PHG_ASM_DATA_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/generated/asm_data/*.c)
file(GLOB PHG_STUB_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/generated/stubs/*.c)
add_executable(pokeheartgold
    game_main_pc.c
    ${PHG_GAME_SOURCES}
    ${PHG_ASM_DATA_SOURCES}
    ${PHG_STUB_SOURCES}
    nitro/misc_data_pc.c
    nitro/online_data_pc.c)
target_link_libraries(pokeheartgold PRIVATE phg_nitro)
# Generated by a custom command declared in cmake/PhgGame.cmake (top level).
set_source_files_properties(${PHG_GENERATED_DIR}/pc_overlay_ids.c PROPERTIES GENERATED TRUE)
add_dependencies(pokeheartgold phg_generated_headers)
add_dependencies(phg_nitro phg_generated_headers)
phg_low_address_link(pokeheartgold)
if(MINGW)
    target_link_options(pokeheartgold PRIVATE -static)
endif()
if(WIN32 AND CMAKE_C_COMPILER_ID MATCHES "Clang")
    # lld (llvm-mingw) only: GNU ld (Linux) and ld64 (macOS) reject these.
    target_link_options(pokeheartgold PRIVATE -Wl,--error-limit=0 -Wl,--no-demangle)
endif()
if(CMAKE_C_COMPILER_ID MATCHES "Clang")
    # Decompiled code: keep the build readable, not pedantic.
    target_compile_options(pokeheartgold PRIVATE -w)
endif()
# The target keeps its name; the file is pokeheartgold or pokesoulsilver.
string(TOLOWER "poke${PHG_GAME_VERSION}" PHG_EXE_NAME)
set_target_properties(pokeheartgold PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}
    OUTPUT_NAME ${PHG_EXE_NAME})

# --- Overlay variables: fresh on every load, as on the DS ---------------------
# Each overlay's sources get their own .data section (Clang pragma), so the
# Nitro layer can snapshot it at start-up and restore it in FS_LoadOverlay*.
# See tools/pc_port/gen_overlay_sections.py.
if(CMAKE_C_COMPILER_ID MATCHES "Clang")
    execute_process(
        COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tools/pc_port/gen_overlay_sections.py
                ${PHG_GENERATED_DIR}
        RESULT_VARIABLE _ovl_result
        OUTPUT_VARIABLE _ovl_output
        OUTPUT_STRIP_TRAILING_WHITESPACE)
    if(NOT _ovl_result EQUAL 0)
        message(FATAL_ERROR "gen_overlay_sections.py failed: ${_ovl_output}")
    endif()
    message(STATUS "Overlay sections: ${_ovl_output}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
        ${PROJECT_SOURCE_DIR}/main.lsf
        ${PROJECT_SOURCE_DIR}/tools/pc_port/gen_overlay_sections.py)
    include(${PHG_GENERATED_DIR}/pc_overlay_sources.cmake)
    target_sources(phg_nitro PRIVATE ${PHG_GENERATED_DIR}/pc_overlay_sections.c)
    target_compile_definitions(phg_nitro PRIVATE PHG_OVERLAY_RESET=1)

    # Soft reset (OS_ResetSystem): every other game variable goes to
    # ".data$main$1". Overlay sources add their own -include after this one.
    # (-include<file> as one token: CMake de-duplicates repeated "-include".)
    set(_phg_main_section -fno-zero-initialized-in-bss "-include${PHG_GENERATED_DIR}/pc_ovl/main.h")
    target_compile_options(pokeheartgold PRIVATE ${_phg_main_section})
    # Game-facing state of the Nitro layer (arenas, IRQ table, alarms, DMA,
    # touch screen...). Not nitro_pc.c / fs_nitro_pc.c: host resources.
    set_property(SOURCE
        nitro/os_nitro_pc.c nitro/os_extra_pc.c nitro/mi_dma_pc.c
        nitro/sdk_asm_pc.c nitro/input_nitro_pc.c nitro/card_nitro_pc.c
        nitro/rtc_nitro_pc.c nitro/mi_nitro_pc.c nitro/nnsys_data_pc.c
        ${PROJECT_SOURCE_DIR}/lib/NitroSDK/src/os/os_alloc.c
        ${PROJECT_SOURCE_DIR}/lib/NitroSDK/src/rtc/rtc_convert_pre.c
        ${PROJECT_SOURCE_DIR}/lib/NitroSDK/src/rtc/rtc_convert_post.c
        ${PROJECT_SOURCE_DIR}/lib/NitroSDK/src/rtc/rtc_convert_date_time_to_second.c
        ${PROJECT_SOURCE_DIR}/lib/NitroSystem/src/fnd_allocator.c
        ${PROJECT_SOURCE_DIR}/lib/NitroSystem/src/fnd_expheap.c
        ${PROJECT_SOURCE_DIR}/lib/NitroSystem/src/fnd_frameheap.c
        ${PROJECT_SOURCE_DIR}/lib/NitroSystem/src/fnd_heapcommon.c
        ${PROJECT_SOURCE_DIR}/lib/NitroSystem/src/fnd_list.c
        ${PROJECT_SOURCE_DIR}/lib/NitroSystem/src/fnd_unitheap.c
        APPEND PROPERTY COMPILE_OPTIONS ${_phg_main_section})
endif()
